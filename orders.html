function checkout(){ if(cart.length===0)return; const total=cart.reduce( (sum,item)=>sum+item.product.price*item.quantity,0 ); const orderNumber="VEL-"+Date.now().toString().slice(-6); // Save this demo order const orders=JSON.parse(localStorage.getItem("velora_orders")||"[]"); orders.unshift({ number:orderNumber, total:total }); localStorage.setItem("velora_orders",JSON.stringify(orders)); document.getElementById("order-number").textContent="Order #"+orderNumber; document.getElementById("order-total").textContent="Order total: $"+total.toFixed(2); closeCart(); document.getElementById("confirmation").classList.add("open"); cart=[]; updateBag(); }
